Что такое цикл?

Цикл позволяет выполнять один и тот же код много раз.

Цикл for

// Вывести числа от 1 до 5
for (let i = 1; i <= 5; i++) {
    console.log(i);
}
// Выведет: 1, 2, 3, 4, 5

let i = 1 - начальное значение

i <= 5 - условие продолжения

i++ - что делать после каждой итерации

Перебор массива

let fruits = ['яблоко', 'банан', 'апельсин'];

for (let i = 0; i < fruits.length; i++) {
    console.log(fruits[i]);
}

Цикл while

let count = 0;

while (count < 5) {
    console.log(count);
    count++;
}
// Выведет: 0, 1, 2, 3, 4

While выполняется ПОКА условие true

Цикл do-while

let num = 0;

do {
    console.log(num);
    num++;
} while (num < 3);

// Выполнится минимум 1 раз, даже если условие false

Break - прервать цикл

for (let i = 0; i < 10; i++) {
    if (i === 5) {
        break; // остановить цикл
    }
    console.log(i);
}
// Выведет: 0, 1, 2, 3, 4

Continue - пропустить итерацию

for (let i = 0; i < 5; i++) {
    if (i === 2) {
        continue; // пропустить эту итерацию
    }
    console.log(i);
}
// Выведет: 0, 1, 3, 4 (пропустили 2)

Вложенные циклы

// Таблица умножения
for (let i = 1; i <= 3; i++) {
    for (let j = 1; j <= 3; j++) {
        console.log(`${i} x ${j} = ${i * j}`);
    }
}

For...of (современный способ)

let colors = ['red', 'green', 'blue'];

for (let color of colors) {
    console.log(color);
}
// Выведет: red, green, blue
