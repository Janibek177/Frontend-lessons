Что такое функция?

Функция - это блок кода, который можно вызывать многократно. Это как рецепт: написал один раз, используешь много раз.

Объявление функции

function sayHello() {
    console.log('Привет!');
}

// Вызов функции
sayHello(); // Выведет: Привет!
sayHello(); // Можно вызывать сколько угодно раз

Функция с параметрами

function greet(name) {
    console.log('Привет, ' + name + '!');
}

greet('Иван'); // Привет, Иван!
greet('Мария'); // Привет, Мария!

Несколько параметров

function add(a, b) {
    console.log(a + b);
}

add(5, 3); // 8
add(10, 20); // 30

Return - возврат значения

function multiply(a, b) {
    return a * b;
}

let result = multiply(5, 3);
console.log(result); // 15

// Можно использовать сразу
console.log(multiply(10, 2)); // 20

return - возвращает значение и завершает функцию

Без return функция вернет undefined

Параметры по умолчанию

function greet(name = 'Гость') {
    console.log('Привет, ' + name);
}

greet('Иван'); // Привет, Иван
greet(); // Привет, Гость

Стрелочные функции (современный способ)

// Обычная функция
function add(a, b) {
    return a + b;
}

// Стрелочная функция
const add = (a, b) => {
    return a + b;
};

// Короткая форма (если одна строка)
const add = (a, b) => a + b;

console.log(add(5, 3)); // 8

Функция в переменной

const sayHi = function() {
    console.log('Привет!');
};

sayHi();

Область видимости (scope)

let globalVar = 'Глобальная';

function test() {
    let localVar = 'Локальная';
    console.log(globalVar); // Работает
    console.log(localVar); // Работает
}

test();
console.log(globalVar); // Работает
console.log(localVar); // Ошибка! localVar не существует здесь

Практический пример

// Калькулятор
function calculator(num1, num2, operation) {
    if (operation === '+') {
        return num1 + num2;
    } else if (operation === '-') {
        return num1 - num2;
    } else if (operation === '*') {
        return num1 * num2;
    } else if (operation === '/') {
        return num1 / num2;
    }
}

console.log(calculator(10, 5, '+')); // 15
console.log(calculator(10, 5, '*')); // 50
