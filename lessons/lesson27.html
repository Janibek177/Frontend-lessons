Что такое события?

События - это действия, которые происходят на веб-странице: клики, движения мыши, нажатия клавиш и т.д.

addEventListener

let button = document.querySelector('.my-button');

button.addEventListener('click', function() {
    console.log('Кнопка нажата!');
});

// Стрелочная функция
button.addEventListener('click', () => {
    console.log('Клик!');
});

Событие Click

<button id="myBtn">Нажми меня</button>

<script>
let btn = document.getElementById('myBtn');
btn.addEventListener('click', () => {
    alert('Привет!');
});
</script>

Event Object (объект события)

button.addEventListener('click', function(event) {
    console.log(event.target); // элемент, на котором произошло событие
    console.log(event.type); // 'click'
    console.log(event.clientX); // координата X мыши
    console.log(event.clientY); // координата Y мыши
});

Популярные события мыши

click - клик мышью

dblclick - двойной клик

mouseover - курсор навелся на элемент

mouseout - курсор ушел с элемента

mousemove - движение мыши

События клавиатуры

document.addEventListener('keydown', (event) => {
    console.log('Нажата клавиша:', event.key);
    if (event.key === 'Enter') {
        console.log('Нажат Enter!');
    }
});

keydown - клавиша нажата

keyup - клавиша отпущена

keypress - клавиша нажата (устарело)

События формы

let input = document.querySelector('input');

// Изменение значения
input.addEventListener('input', (e) => {
    console.log(e.target.value);
});

// Отправка формы
let form = document.querySelector('form');
form.addEventListener('submit', (e) => {
    e.preventDefault(); // отменить отправку
    console.log('Форма отправлена!');
});

Удаление обработчика

function handleClick() {
    console.log('Клик!');
}

button.addEventListener('click', handleClick);

// Удалить обработчик
button.removeEventListener('click', handleClick);

Event delegation (делегирование)

// Вместо добавления обработчика каждой кнопке
document.querySelector('.container').addEventListener('click', (e) => {
    if (e.target.tagName === 'BUTTON') {
        console.log('Кнопка нажата:', e.target.textContent);
    }
});
