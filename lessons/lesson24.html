Что такое массив?

Массив - это список значений. Как коробка с пронумерованными ячейками.

Создание массива

let fruits = ['яблоко', 'банан', 'апельсин'];
let numbers = [1, 2, 3, 4, 5];
let mixed = ['текст', 42, true, null];

Доступ к элементам (индексация с 0)

let fruits = ['яблоко', 'банан', 'апельсин'];

console.log(fruits[0]); // 'яблоко'
console.log(fruits[1]); // 'банан'
console.log(fruits[2]); // 'апельсин'

// Изменить элемент
fruits[1] = 'груша';
console.log(fruits); // ['яблоко', 'груша', 'апельсин']

Свойство length

let fruits = ['яблоко', 'банан', 'апельсин'];
console.log(fruits.length); // 3

// Последний элемент
console.log(fruits[fruits.length - 1]); // 'апельсин'

Добавление элементов

let fruits = ['яблоко'];

// push - добавить в конец
fruits.push('банан');
console.log(fruits); // ['яблоко', 'банан']

// unshift - добавить в начало
fruits.unshift('груша');
console.log(fruits); // ['груша', 'яблоко', 'банан']

Удаление элементов

let fruits = ['яблоко', 'банан', 'апельсин'];

// pop - удалить последний
let last = fruits.pop();
console.log(last); // 'апельсин'
console.log(fruits); // ['яблоко', 'банан']

// shift - удалить первый
let first = fruits.shift();
console.log(first); // 'яблоко'
console.log(fruits); // ['банан']

splice - удалить/вставить в середину

let arr = [1, 2, 3, 4, 5];

// Удалить 2 элемента начиная с индекса 1
arr.splice(1, 2);
console.log(arr); // [1, 4, 5]

// Вставить элементы
arr.splice(1, 0, 'a', 'b');
console.log(arr); // [1, 'a', 'b', 4, 5]

slice - скопировать часть массива

let arr = [1, 2, 3, 4, 5];
let part = arr.slice(1, 4);
console.log(part); // [2, 3, 4]
console.log(arr); // [1, 2, 3, 4, 5] (не изменился)

forEach - перебор массива

let fruits = ['яблоко', 'банан', 'апельсин'];

fruits.forEach(function(fruit) {
    console.log(fruit);
});

// Стрелочная функция
fruits.forEach(fruit => console.log(fruit));

map - создать новый массив

let numbers = [1, 2, 3, 4];
let doubled = numbers.map(num => num * 2);
console.log(doubled); // [2, 4, 6, 8]

filter - отфильтровать массив

let numbers = [1, 2, 3, 4, 5, 6];
let even = numbers.filter(num => num % 2 === 0);
console.log(even); // [2, 4, 6]

find - найти элемент

let numbers = [1, 5, 10, 15];
let result = numbers.find(num => num > 7);
console.log(result); // 10

includes - проверить наличие

let fruits = ['яблоко', 'банан'];
console.log(fruits.includes('банан')); // true
console.log(fruits.includes('груша')); // false

2. Добавьте новый фильм в конец и в начало

3. Создайте массив чисел от 1 до 10 и отфильтруйте только четные

4. Используйте map чтобы удвоить все числа в массиве
