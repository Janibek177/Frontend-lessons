Арифметические операторы

let a = 10;
let b = 3;

console.log(a + b); // 13 (сложение)
console.log(a - b); // 7 (вычитание)
console.log(a * b); // 30 (умножение)
console.log(a / b); // 3.333... (деление)
console.log(a % b); // 1 (остаток от деления)
console.log(a ** b); // 1000 (возведение в степень)

Инкремент и декремент

let count = 5;
count++; // count = 6 (увеличение на 1)
count--; // count = 5 (уменьшение на 1)

count += 10; // count = 15 (count = count + 10)
count -= 5; // count = 10 (count = count - 5)
count *= 2; // count = 20 (count = count * 2)
count /= 4; // count = 5 (count = count / 4)

Операторы сравнения

5 == '5' // true (равно, не строгое)
5 === '5' // false (равно, строгое - рекомендуется!)
5 != 3 // true (не равно)
5 !== '5' // true (не равно строгое)
5 > 3 // true (больше)
5 < 3 // false (меньше)
5 >= 5 // true (больше или равно)
3 <= 5 // true (меньше или равно)

Важно! Всегда используй === вместо ==

=== проверяет и значение, и тип данных

Логические операторы

// && (И) - оба условия должны быть true
let age = 20;
let hasLicense = true;
if (age >= 18 && hasLicense) {
    console.log('Можете водить');
}

// || (ИЛИ) - хотя бы одно условие true
let isWeekend = true;
let isHoliday = false;
if (isWeekend || isHoliday) {
    console.log('Выходной!');
}

// ! (НЕ) - инверсия
let isRaining = false;
if (!isRaining) {
    console.log('Можно гулять');
}

Тернарный оператор

// Краткая форма if-else
let age = 20;
let status = age >= 18 ? 'взрослый' : 'ребенок';
console.log(status); // 'взрослый'

// Формула: условие ? значение_если_true : значение_если_false

Оператор typeof

console.log(typeof 42); // 'number'
console.log(typeof 'hello'); // 'string'
console.log(typeof true); // 'boolean'
console.log(typeof undefined); // 'undefined'
console.log(typeof null); // 'object' (это баг в JavaScript!)

Конкатенация строк

let firstName = 'Иван';
let lastName = 'Петров';
let fullName = firstName + ' ' + lastName;
console.log(fullName); // 'Иван Петров'

// Шаблонные строки (лучший способ)
let greeting = `Привет, ${firstName} ${lastName}!`;
console.log(greeting); // 'Привет, Иван Петров!'
