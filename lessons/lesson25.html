Что такое объект?

Объект - это коллекция свойств (ключ: значение). Как словарь или картотека.

Создание объекта

let person = {
    name: 'Иван',
    age: 25,
    city: 'Москва'
};

Доступ к свойствам

let person = {
    name: 'Иван',
    age: 25
};

// Точечная нотация
console.log(person.name); // 'Иван'

// Квадратные скобки
console.log(person['age']); // 25

// Изменить свойство
person.age = 26;

// Добавить новое свойство
person.city = 'Москва';

Удаление свойств

let person = { name: 'Иван', age: 25 };
delete person.age;
console.log(person); // { name: 'Иван' }

Методы объекта

let person = {
    name: 'Иван',
    age: 25,
    greet: function() {
        console.log('Привет!');
    }
};

person.greet(); // 'Привет!'

This - ссылка на объект

let person = {
    name: 'Иван',
    age: 25,
    introduce: function() {
        console.log(`Меня зовут ${this.name}, мне ${this.age} лет`);
    }
};

person.introduce(); // 'Меня зовут Иван, мне 25 лет'

Перебор свойств (for...in)

let person = {
    name: 'Иван',
    age: 25,
    city: 'Москва'
};

for (let key in person) {
    console.log(key + ': ' + person[key]);
}
// name: Иван
// age: 25
// city: Москва

Object.keys, values, entries

let person = { name: 'Иван', age: 25 };

// Получить все ключи
console.log(Object.keys(person)); // ['name', 'age']

// Получить все значения
console.log(Object.values(person)); // ['Иван', 25]

// Получить пары [ключ, значение]
console.log(Object.entries(person)); // [['name', 'Иван'], ['age', 25]]

Вложенные объекты

let user = {
    name: 'Иван',
    address: {
        city: 'Москва',
        street: 'Ленина',
        house: 10
    }
};

console.log(user.address.city); // 'Москва'

Копирование объекта

let person = { name: 'Иван', age: 25 };

// Поверхностное копирование
let copy = { ...person };

// Или
let copy2 = Object.assign({}, person);

Проверка наличия свойства

let person = { name: 'Иван' };

console.log('name' in person); // true
console.log('age' in person); // false
